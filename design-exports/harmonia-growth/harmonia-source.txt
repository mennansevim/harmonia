<script>
  (() => {
    const root = document.getElementById('harmonia-growth-preview');
    const phone = root.querySelector('.hg-phone');
    const live = root.querySelector('.hg-live');
    const names = { today: 'Bugün', train: 'Çalışma', result: 'Sonuç', premium: 'Üyelik' };

    function showScreen(name) {
      root.querySelectorAll('[data-screen]').forEach((screen) => {
        const active = screen.dataset.screen === name;
        screen.hidden = !active;
        screen.classList.toggle('is-active', active);
      });
      root.querySelectorAll('[data-nav]').forEach((button) => {
        button.classList.toggle('is-active', button.dataset.nav === name);
      });
      live.textContent = `${names[name]} ekranı açıldı`;
    }

    root.querySelectorAll('[data-nav]').forEach((button) => button.addEventListener('click', () => showScreen(button.dataset.nav)));
    root.querySelectorAll('[data-go]').forEach((button) => button.addEventListener('click', () => showScreen(button.dataset.go)));

    const breath = root.querySelector('#hg-breath-toggle');
    let paused = false;
    breath.addEventListener('click', () => {
      paused = !paused;
      breath.innerHTML = paused ? 'Devam et <i data-lucide="play" aria-hidden="true"></i>' : 'Duraklat <i data-lucide="pause" aria-hidden="true"></i>';
      live.textContent = paused ? 'Nefes döngüsü duraklatıldı' : 'Nefes döngüsü devam ediyor';
      if (globalThis.lucide) lucide.createIcons({ attrs: { width: 16, height: 16 } });
    });

    root.querySelectorAll('[data-plan]').forEach((button) => button.addEventListener('click', () => {
      const plan = button.dataset.plan;
      root.querySelectorAll('[data-plan]').forEach((tab) => {
        const selected = tab.dataset.plan === plan;
        tab.classList.toggle('is-selected', selected);
        tab.setAttribute('aria-selected', String(selected));
      });
      root.querySelectorAll('[data-plan-panel]').forEach((panel) => { panel.hidden = panel.dataset.planPanel !== plan; });
      live.textContent = `${plan === 'plus' ? 'Plus' : 'Premium'} planı gösteriliyor`;
    }));

    const design = { accent: '#5b46f5', radius: 30 };
    function renderDesign() {
      phone.style.setProperty('--hg-accent', design.accent);
      phone.style.borderRadius = `${design.radius}px`;
    }
    renderDesign();
    if (globalThis.Tweak) {
      const tweak = new Tweak({ container: phone, onChange: renderDesign });
      tweak.addColorPicker(design, 'accent', { label: 'Vurgu rengi', reference: '--hg-accent' });
      tweak.addSlider(design, 'radius', { label: 'Dış köşe', min: 12, max: 40, unit: 'px', reference: 'phone.radius' });
    }
  })();
</script>
